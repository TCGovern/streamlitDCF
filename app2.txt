def pick(df, *labels, default=np.nan):
    """Return the first matching column label, else a default."""
    for label in labels:
        if label in df.columns:
            return df[label]
    return default


def col(df, *labels):
    """pick() but always returns a Series (NaN-filled if nothing matched)."""
    r = pick(df, *labels)
    if isinstance(r, pd.Series):
        return r
    return pd.Series(np.nan, index=df.index)


def last(series):
    s = series.dropna()
    return float(s.iloc[-1]) if len(s) else np.nan


def num(x, default=np.nan):
    try:
        v = float(x)
        return v if np.isfinite(v) else default
    except (TypeError, ValueError):
        return default


def money(x, d=2):
    return "N/A" if not np.isfinite(num(x)) else f"${x:,.{d}f}"


def big(x):
    x = num(x)
    if not np.isfinite(x):
        return "N/A"
    for unit, div in (("T", 1e12), ("B", 1e9), ("M", 1e6)):
        if abs(x) >= div:
            return f"${x / div:,.2f}{unit}"
    return f"${x:,.0f}"


def pct(x, d=1):
    return "N/A" if not np.isfinite(num(x)) else f"{x * 100:.{d}f}%"

@st.cache_data(show_spinner=False, ttl=3600)
def load_ticker(ticker, force_cache=False):
    """Returns (income, balance, cashflow, info, source_message)."""
    paths = {k: f"{CACHE_DIR}/{ticker}_{k}.csv" for k in ("income", "balance", "cashflow")}
    info_path = f"{CACHE_DIR}/{ticker}_info.json"

    def from_cache():
        frames = {k: pd.read_csv(p, index_col=0, parse_dates=True).sort_index() for k, p in paths.items()}
        with open(info_path) as f:
            info = json.load(f)
        return frames["income"], frames["balance"], frames["cashflow"], info

    if force_cache:
        try:
            return (*from_cache(), "Loaded from cached CSV files.")
        except Exception as e:
            return None, None, None, {}, f"No cache available for {ticker} ({e})."

    try:
        stock = yf.Ticker(ticker)
        inc = stock.income_stmt.T.sort_index()
        bal = stock.balance_sheet.T.sort_index()
        cf = stock.cashflow.T.sort_index()
        info = stock.info or {}
        if inc.empty or bal.empty or cf.empty:
            raise ValueError("yfinance returned empty statements")
        for df, k in ((inc, "income"), (bal, "balance"), (cf, "cashflow")):
            df.to_csv(paths[k])
        with open(info_path, "w") as f:
            json.dump(info, f, default=str)
        return inc, bal, cf, info, "Live data from yfinance (saved to cache)."
    except Exception as e:
        try:
            return (*from_cache(), f"yfinance failed ({e}); using cached CSV.")
        except Exception:
            return None, None, None, {}, f"yfinance failed ({e}) and no cache exists."

def dcf(base_fcf, growth, term_g, wacc, years, cash, debt, shares):
    """Returns dict; value is NaN if g >= r."""
    if term_g >= wacc or not np.isfinite(base_fcf):
        return dict(intrinsic=np.nan, enterprise=np.nan, tv_pct=np.nan, pv=None, projected=None)
    t = np.arange(1, years + 1)
    projected = base_fcf * (1 + growth) ** t
    pv = projected / (1 + wacc) ** t
    terminal = projected[-1] * (1 + term_g) / (wacc - term_g)
    pv_terminal = terminal / (1 + wacc) ** years
    enterprise = pv.sum() + pv_terminal
    equity = enterprise + cash - debt
    return dict(
        intrinsic=equity / shares if shares else np.nan,
        enterprise=enterprise,
        equity=equity,
        tv_pct=pv_terminal / enterprise if enterprise else np.nan,
        pv=pv,
        projected=projected,
        pv_terminal=pv_terminal,
    )


def reverse_dcf(price, base_fcf, term_g, wacc, years, cash, debt, shares):
    """Bisection: forecast growth that makes intrinsic == market price."""
    lo, hi = -0.30, 0.60
    f = lambda g: dcf(base_fcf, g, term_g, wacc, years, cash, debt, shares)["intrinsic"] - price
    flo, fhi = f(lo), f(hi)
    if not (np.isfinite(flo) and np.isfinite(fhi)) or flo * fhi > 0:
        return np.nan
    for _ in range(80):
        mid = (lo + hi) / 2
        if f(lo) * f(mid) <= 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def heatmap(z, xs, ys, xtitle, ytitle, title, price_ref):
    zt = np.array(z, dtype=float)
    text = [[("N/A" if not np.isfinite(v) else f"${v:,.0f}") for v in row] for row in zt]
    fig = go.Figure(go.Heatmap(
        z=zt, x=[f"{x:.2%}" for x in xs], y=[f"{y:.2%}" for y in ys],
        text=text, texttemplate="%{text}", colorscale="RdYlGn",
        zmid=price_ref if np.isfinite(price_ref) else None,
        colorbar=dict(title="$ / share"),
    ))
    ci, cj = len(ys) // 2, len(xs) // 2
    fig.add_shape(type="rect", x0=cj - 0.5, x1=cj + 0.5, y0=ci - 0.5, y1=ci + 0.5,
                  line=dict(color="black", width=4))
    fig.update_layout(title=title, xaxis_title=xtitle, yaxis_title=ytitle, yaxis_autorange="reversed")
    return fig

        bars.append(("DCF (sensitivity range)", np.nanmin(sens_vals), np.nanmax(sens_vals)))
    for m, (lo, hi) in comps_ranges.items():
        if np.isfinite(lo) and np.isfinite(hi):
            bars.append((f"Comps: {m}", min(lo, hi), max(lo, hi)))
    lo52, hi52 = num(info.get("fiftyTwoWeekLow")), num(info.get("fiftyTwoWeekHigh"))
    if np.isfinite(lo52) and np.isfinite(hi52):
        bars.append(("52-week range", lo52, hi52))
    if bars:
        fig = go.Figure()
        for label, lo, hi in bars:
            fig.add_bar(y=[label], x=[hi - lo], base=[lo], orientation="h", name=label,
                        text=[f"${lo:,.0f} - ${hi:,.0f}"], textposition="inside", showlegend=False)
        if np.isfinite(price):
            fig.add_vline(x=price, line_dash="dash", line_color="black",
                          annotation_text=f"Price ${price:,.2f}")
        fig.update_layout(title="Valuation range summary ($ per share)", xaxis_title="$ per share",
                          yaxis_title="Method", yaxis_autorange="reversed")
        st.plotly_chart(fig, use_container_width=True)
    else:
        st.info("Not enough data for a football field yet.")

    st.subheader("Reverse DCF")
    if np.isfinite(price) and np.isfinite(base_fcf) and base_fcf > 0 and term_g < wacc:
        g_imp = reverse_dcf(price, base_fcf, term_g, wacc, years, cash, debt, shares)
        if np.isfinite(g_imp):
            hist = fcf_clean.pct_change().dropna()
            hist_txt = f" Historical FCF growth in the data averaged {hist.mean():.1%}." if len(hist) else ""
            st.metric(f"FCF growth implied by ${price:,.2f} over {years} years", pct(g_imp))
            st.write(f"At WACC {pct(wacc, 2)} and terminal growth {pct(term_g, 2)}, the market price requires "
                     f"{pct(g_imp)} annual FCF growth.{hist_txt} Judge whether that is believable.")
        else:
            st.warning("No growth rate between -30% and 60% reproduces the market price.")
    else:
        st.info("Reverse DCF needs a valid price, positive base FCF and terminal growth below WACC.")
